export interface CVAnalysisEvent {
  eventId: string;
    candidateId: string;
      timestamp: Date;
        cvText: string;
          cvLengthChars: number;
            
              // Score antes/depois
                scoreBefore: number;
                  scoreAfter: number;
                    scoreImprovement: number;
                      
                        // Mudanças aplicadas
                          changesApplied: {
                              category: 'bullet_points' | 'keywords' | 'structure' | 'summary';
                                  count: number;
                                    }[];
                                      
                                        // Candidato aceita sugestão?
                                          suggestionsAccepted: number;
                                            suggestionsRejected: number;
                                              acceptanceRate: number; // %
                                                
                                                  // Contexto
                                                    industry: string;
                                                      seniority: 'JUNIOR' | 'PLENO' | 'SENIOR' | 'LEAD' | 'MANAGER';
                                                        location: string;
                                                          desiredRole: string;
                                                            
                                                              // (Futuro) Feedback de resultado
                                                                timeToHire?: number; // dias
                                                                  salaryRange?: string;
                                                                    hired?: boolean;
                                                                      feedback?: string;
                                                                        
                                                                          // LGPD consent
                                                                            consentGiven: boolean;
                                                                              consentVersion: string;
                                                                              }

                                                                              export class AnalyticsService {
                                                                                private events: CVAnalysisEvent[] = [];

                                                                                  logAnalysis(event: CVAnalysisEvent): void {
                                                                                      // Save to localStorage (MVP)
                                                                                          this.events.push(event);
                                                                                              localStorage.setItem('recruta_cv_events', JSON.stringify(this.events));
                                                                                                  
                                                                                                      // Log para debug
                                                                                                          console.log('[Moat] CV Analysis Event:', {
                                                                                                                candidate: event.candidateId,
                                                                                                                      improvement: event.scoreImprovement,
                                                                                                                            acceptanceRate: event.acceptanceRate,
                                                                                                                                  industry: event.industry
                                                                                                                                      });
                                                                                                                                        }

                                                                                                                                          getEventsForAnalysis(): CVAnalysisEvent[] {
                                                                                                                                              return this.events;
                                                                                                                                                }

                                                                                                                                                  getAggregatedInsights(industry?: string) {
                                                                                                                                                      const filtered = industry
                                                                                                                                                            ? this.events.filter(e => e.industry === industry)
                                                                                                                                                                  : this.events;

                                                                                                                                                                      return {
                                                                                                                                                                            totalAnalyzed: filtered.length,
                                                                                                                                                                                  averageScoreImprovement: 
                                                                                                                                                                                          filtered.reduce((sum, e) => sum + e.scoreImprovement, 0) / filtered.length,
                                                                                                                                                                                                averageAcceptanceRate: 
                                                                                                                                                                                                        filtered.reduce((sum, e) => sum + e.acceptanceRate, 0) / filtered.length,
                                                                                                                                                                                                              topChanges: this.getTopChanges(filtered),
                                                                                                                                                                                                                    byIndustry: this.groupByIndustry(filtered),
                                                                                                                                                                                                                        };
                                                                                                                                                                                                                          }

                                                                                                                                                                                                                            private getTopChanges(events: CVAnalysisEvent[]) {
                                                                                                                                                                                                                                // Qual tipo de mudança tem maior impact?
                                                                                                                                                                                                                                    return events
                                                                                                                                                                                                                                          .flatMap(e => e.changesApplied)
                                                                                                                                                                                                                                                .reduce((acc, change) => {
                                                                                                                                                                                                                                                        acc[change.category] = (acc[change.category] || 0) + change.count;
                                                                                                                                                                                                                                                                return acc;
                                                                                                                                                                                                                                                                      }, {} as Record<string, number>);
                                                                                                                                                                                                                                                                        }

                                                                                                                                                                                                                                                                          private groupByIndustry(events: CVAnalysisEvent[]) {
                                                                                                                                                                                                                                                                              return events.reduce((acc, e) => {
                                                                                                                                                                                                                                                                                    if (!acc[e.industry]) {
                                                                                                                                                                                                                                                                                            acc[e.industry] = { count: 0, avgImprovement: 0 };
                                                                                                                                                                                                                                                                                                  }
                                                                                                                                                                                                                                                                                                        acc[e.industry].count++;
                                                                                                                                                                                                                                                                                                              acc[e.industry].avgImprovement += e.scoreImprovement;
                                                                                                                                                                                                                                                                                                                    return acc;
                                                                                                                                                                                                                                                                                                                        }, {} as Record<string, any>);
                                                                                                                                                                                                                                                                                                                          }
                                                                                                                                                                                                                                                                                                                          }

                                                                                                                                                                                                                                                                                                                          export const analyticsService = new AnalyticsService();
